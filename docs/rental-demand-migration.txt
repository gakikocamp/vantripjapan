-- VanTripJapan rental demand migration
-- ====================================
-- Target binding/database: CUSTOMERS_DB / vtj-customers
--
-- This migration is intentionally kept as a tracked .txt file because this public
-- repository ignores *.sql files. Review it, then pass this file to Wrangler's
-- --file option during the approved deployment procedure. Do not apply it directly
-- to production outside that procedure.

CREATE TABLE IF NOT EXISTS rental_searches (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  searched_at TEXT NOT NULL DEFAULT (datetime('now')),
  pickup_date TEXT NOT NULL,
  return_date TEXT NOT NULL,
  rental_days INTEGER NOT NULL,
  lead_days INTEGER NOT NULL,
  guests INTEGER NOT NULL,
  compatible_count INTEGER NOT NULL,
  available_count INTEGER,
  availability_complete INTEGER NOT NULL DEFAULT 0,
  language TEXT NOT NULL DEFAULT 'en',
  landing_path TEXT NOT NULL DEFAULT '',
  referrer_host TEXT NOT NULL DEFAULT '',
  utm_source TEXT NOT NULL DEFAULT '',
  utm_medium TEXT NOT NULL DEFAULT '',
  utm_campaign TEXT NOT NULL DEFAULT '',
  country TEXT NOT NULL DEFAULT '',
  dedupe_key TEXT NOT NULL UNIQUE
);

CREATE INDEX IF NOT EXISTS idx_rental_searches_pickup
  ON rental_searches(pickup_date, return_date);
CREATE INDEX IF NOT EXISTS idx_rental_searches_created
  ON rental_searches(searched_at);
